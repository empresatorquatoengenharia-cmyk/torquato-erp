// Substituir a listagem estática por requisição real ao Supabase async function carregarContas() { const obraFiltro = document.getElementById('filtro-obra').value; const statusFiltro = document.getElementById('filtro-status').value; const tbody = document.getElementById('tabela-contas-corpo'); let query = supabaseClient.from('financeiro_contas_pagar').select('*'); if (obraFiltro !== 'Todos') { query = query.eq('obra', obraFiltro); } if (statusFiltro !== 'Todos') { query = query.eq('status', statusFiltro); } const { data, error } = await query; if (error) { tbody.innerHTML = `Erro ao carregar dados: ${error.message}`; return; } const lista = data || []; if (lista.length === 0) { tbody.innerHTML = `Nenhuma conta encontrada.`; return; } tbody.innerHTML = lista.map(c => ` #${c.codigo} ${c.nfe || '-'} ${c.beneficiario} R$ ${(c.valor || 0).toLocaleString('pt-BR', {minimumFractionDigits: 2})} ${c.vencimento} ${c.pago_em || '-'} ${c.caixa || '-'} ${c.etiqueta} `).join(''); } // Efetivar pagamento gravando diretamente na nuvem async function efetivarPagamentoReal(id, banco, dataPag) { const { error } = await supabaseClient .from('financeiro_contas_pagar') .update({ status: 'Paga', pago_em: dataPag, caixa: banco, etiqueta: 'Paga' }) .eq('id', id); if (error) { alert('Erro ao efetivar pagamento: ' + error.message); } else { alert('Pagamento registado e sincronizado com sucesso no Supabase!'); carregarContas(); fecharModalDetalhes(); } }

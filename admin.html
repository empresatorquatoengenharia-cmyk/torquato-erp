async function salvarMatriz() { const razao = document.getElementById('razao_social')?.value || ''; const cnpj = document.getElementById('cnpj')?.value || ''; const ie = document.getElementById('inscricao_estadual')?.value || ''; const im = document.getElementById('inscricao_municipal')?.value || ''; const cep = document.getElementById('cep')?.value || ''; const estado = document.getElementById('estado')?.value || ''; const pais = document.getElementById('pais')?.value || ''; try { // Tenta upsert (inserir ou atualizar ID 1) const { data, error } = await supabaseClient .from('empresa_matriz') .upsert([ { id: 1, razao_social: razao, cnpj: cnpj, inscricao_estadual: ie, inscricao_municipal: im, cep: cep, estado: estado, pais: pais } ]); if (error) { console.error('Erro detalhado do Supabase:', error); alert(`Erro ao salvar: ${error.message} (${error.details || error.hint || ''})`); } else { alert('Dados da Matriz salvos com sucesso!'); } } catch (err) { console.error('Erro inesperado:', err); alert('Erro ao conectar ao banco de dados: ' + err.message); } }
